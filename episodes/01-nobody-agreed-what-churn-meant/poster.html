CREATE OR REPLACE TABLE analytics.gold.churn_label AS
SELECT subscription_id, billing_cycle_start, market,
       CASE WHEN end_reason = 'subscriber_cancelled'
            THEN 1 ELSE 0 END AS churned_voluntary
FROM   analytics.silver.billing_cycles
WHERE  paid_month_number IN (1,2)        -- who counts
  AND  end_reason <> 'payment_failed'   -- what's excluded
  AND  renewal_window_closed = TRUE;  -- when it's known

COMMENT ON TABLE analytics.gold.churn_label IS
  'Voluntary churn, paid months 1-2. One row per billing cycle.';
The contract is the table. And the roll-up rule: sum, then divide — never average the market rates.
